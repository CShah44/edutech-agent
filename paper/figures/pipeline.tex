    input/.style={

        draw,

    },
    output/.style={
        rectangle,

% Nodes
\node[input] (question) {Question};

\node[agent, below=of question] (breakdown)
    {Breakdown Agent};

\node[agent, below left=1.8cm and 2.8cm of breakdown] (reasoning)
    {Reasoning Agent};

\node[agent, below right=1.8cm and 2.8cm of breakdown] (scientific)
    {Scientific Agent};

\node[agent, below=2.1cm of breakdown] (synthesis)
    {Synthesis Agent};

\node[agent, below=1.5cm of synthesis] (creative)
    {Creative Agent};

\node[output, below=1.5cm of creative] (answer)
    {ELI5 Explanation};

% Connections
\draw[arrow] (question) -- (breakdown);

\draw[arrow]
    (breakdown.south west) --
    node[midway, above, sloped] {Reasoning Points}
    (reasoning.north);

\draw[arrow]
    (breakdown.south east) --
    node[midway, above, sloped] {Search Queries}
    (scientific.north);

\draw[arrow]
    (reasoning.east) --
    node[midway, above] {Logical Analysis}
    (synthesis.west);

\draw[arrow]
    (scientific.west) --
    node[midway, above] {Grounded Facts}
    (synthesis.east);

\draw[arrow]
    (synthesis) --
    node[midway, right] {Curated Points + Strategy}
    (creative);

\draw[arrow] (creative) -- (answer);

\end{tikzpicture}

\caption{
Multi-agent architecture for ELI5 explanation generation.
The breakdown agent decomposes questions into reasoning points and search queries.
The reasoning and scientific agents operate in parallel, producing logical analyses and grounded facts.
The synthesis agent integrates these outputs and selects an adaptive content strategy.
The creative agent transforms curated points into accessible explanations.
}
